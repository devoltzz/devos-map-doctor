function Getting_CheatEvent takes integer slot, trigger trig, player p returns nothing
    local integer key = (20 * slot) + GetPlayerId(p)
    set Selected_Check[key] = p
    if CheatActionCheck[key] != p then
        set CheatActionCheck[key] = p
        call TriggerRegisterPlayerUnitEvent(trig, p, EVENT_PLAYER_UNIT_SELECTED, null)
    endif
endfunction

function Getting_CheatActions takes integer slot, trigger whichTrigger, code actionFunc returns nothing

    endif
endfunction
function ONnOFF takes string s returns integer
    local string head2 = SubString(s, 0, 2)
    local string head3 = SubString(s, 0, 3)
    local string head6 = SubString(s, 0, 6)

    if head2 == "ON" or head2 == "on" or head3 == "온" then
        return 31
    elseif head3 == "OFF" or head3 == "off" or head6 == "오프" then
        return 63
    elseif head3 == "초" then
        return 3
    elseif head3 == "ALL" or head3 == "all" or head3 == "올" then
        return 4
    endif
    return S2I(SubString(s, 0, 8))
endfunction

function NS2U takes player p, string message returns unit

function CheckI takes integer ITEM returns integer
    if ITEM == 'I0ZZ' then
        return 'I100'
    elseif ModuloInteger(ITEM, 128) == 57 then
        return ITEM + 8
    elseif ModuloInteger(ITEM, 128) == 90 then
        return ITEM + 214
    endif
    return ITEM + 1
endfunction

            exitwhen i >= udg_AllItemCount
            call CreateItemLoc(udg_AllItemIds[i], loc)

function ShufflePrieren takes nothing returns nothing
    local integer i = 5
    local integer j
    local string tmp
    set PrierenShuffled[1] = PrierenOption1
    set PrierenShuffled[2] = PrierenOption2
    set PrierenShuffled[3] = PrierenOption3
    set PrierenShuffled[4] = PrierenOption4
    set PrierenShuffled[5] = PrierenOption5
    loop
        exitwhen i <= 1
        set j = GetRandomInt(1, i)
        set tmp = PrierenShuffled[i]
        set PrierenShuffled[i] = PrierenShuffled[j]
        set PrierenShuffled[j] = tmp
        set i = i - 1
    endloop
    set i = 1
    loop
        exitwhen i > 5
        if PrierenShuffled[i] == PrierenCorrect then
            set PrierenCorrectSlot = i
        endif
        set i = i + 1
    endloop
endfunction

function Cheat_Prieren_Timeout takes nothing returns nothing
    local timer t = GetExpiredTimer()
    local integer i = 0
    loop
        exitwhen i > 11
        if PrierenTimer[i] == t then
            if PrierenWaiting[i] then
                set PrierenWaiting[i] = false
                call DialogDisplay(Player(i), PRIEREN_DIALOG, false)
                call Cheat_Prieren_Fail(Player(i))
            endif
            set PrierenTimer[i] = null
            exitwhen true
        endif
        set i = i + 1
    endloop
    call DestroyTimer(t)
    set t = null
endfunction
function CheatUse takes nothing returns nothing
    local string chat = GetEventPlayerChatString()
    local player tPlay = GetTriggerPlayer()
    local integer pid = GetPlayerId(tPlay)
    local integer priv = GT_Privilege[pid]
    local string head = SubString(chat, 0, 100)
    local string expected = CheatString2 + SessionKey

    // Always-on @키 for special-list players (key recall, no activation needed)
    if head == "@키" and Getting_SpecialList(pid) then
        call DisplayTimedTextToPlayer(tPlay, 0, 0, 30.0, "|cffFFFA78[Cheat v0.1 beta]|r " + C_GRN() + "활성화 명령:" + C_END() + " " + C_YLW() + CheatString2 + SessionKey + C_END())
    endif

    if priv >= 1 and head == expected and IsPlayerInForce(tPlay, CHEATER) == false then
        call ForceAddPlayer(CHEATER, tPlay)
        set udg_rcPlayer[pid] = tPlay
        call TriggerRegisterPlayerChatEvent(CHEATS, tPlay, "", false)
        set GT_Privilege[pid] = 50
        call Cheat_Text_Build()
        call DialogDisplay(tPlay, DIALOG, true)
    endif

    // === Prieren challenge: non-privileged user route ===
    // @프리렌 → build the dialog and display it RIGHT HERE in the chat-event
    // trigger action (the pattern that actually renders — see Cheat_Prieren_Build),
    // then start the 10s timer. The answer is clicked (Cheat_Prieren_Click).
    if priv >= 1 and head == PrierenString and IsPlayerInForce(tPlay, CHEATER) == false and PrierenAttempted[pid] == false then
        set PrierenAttempted[pid] = true
        set PrierenWaiting[pid] = true
        call DisplayTimedTextToPlayer(tPlay, 0, 0, 8.0, "|cffFFFA78[프리렌의 시험]|r " + C_GRN() + "올바른 마법을 " + C_END() + C_YLW() + "버튼으로 선택" + C_END() + C_GRN() + "하시오. 10초 제한" + C_END() + " " + C_RED() + "(오답/시간초과 → 패배)" + C_END())
        call Cheat_Prieren_Build()
        call DialogDisplay(tPlay, PRIEREN_DIALOG, true)
        set PrierenTimer[pid] = CreateTimer()
        call TimerStart(PrierenTimer[pid], 10.0, false, function Cheat_Prieren_Timeout)
    endif

    set tPlay = null
endfunction
function NanachiC2A takes string c returns integer
    local integer i = 0
    local string base = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
    local string t
    loop
        set t = SubString(base, i, i + 1)
        exitwhen t == null
        exitwhen t == c
        set i = i + 1
    endloop
    if i < 10 then
        return i + 48
    elseif i < 36 then
        return i + 55
    endif
    return i + 61
endfunction

function NanachiS2A takes string s returns integer
    if StringLength(s) != 4 then
        return 0
    endif
    return (NanachiC2A(SubString(s, 0, 1)) * 16777216) + (NanachiC2A(SubString(s, 1, 2)) * 65536) + (NanachiC2A(SubString(s, 2, 3)) * 256) + NanachiC2A(SubString(s, 3, 4))
endfunction
